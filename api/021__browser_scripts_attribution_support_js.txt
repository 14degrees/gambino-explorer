function getQueryVariable(searchString, variable) {
    var query = searchString.substring(1);
    var vars = query.split("&");
    for (var i=0; i < vars.length;i++) {
        var pair = vars[i].split("=");
        if(pair[0] == variable) {
            return pair[1];
        }
    }
    return(false);
}

function tryCallPikoyaPixel(uri) {
    var searchIndex = uri.indexOf("?");
    if (searchIndex != -1) {
        var searchString = uri.substring(searchIndex);

        var s2 = getQueryVariable(searchString, "s2");
        if (!s2){
            s2 = getQueryVariable(searchString, "S2");
        }

        if (!s2) {
            return false;
        }

        console.log("Found Picoya s2 parameter: " + s2);

        var img = document.createElement("img");
        img.src = "https://traffic.fstsrv.com/conversions?key=" + s2;
        img.style.display = "none";
        var src = document.getElementById("bod");
        src.appendChild(img);

        console.log("Called Picoya Pixel " + img.src);
        return true;
    }
    return false;
}

function processReferrerUri() {
    var clickId = getQueryVariable(window.location.search, "cid");

    if (!clickId) {
        return;
    }

    const client = new XMLHttpRequest();
    const url='https://i.gambinoslot.com/api/';

    client.open("GET", url);

    client.setRequestHeader("X-GNBCLID", clickId);
    client.setRequestHeader("Accept", "*/*");

    client.onload = (e) => {
        var uri = client.responseText;
        console.log("https://i.gambinoslot.com/api response: " + uri);

        if (tryCallPikoyaPixel(uri)){
            return;
        }
    }

    client.send();
}
